ООП:

#include <iostream>
#include <cmath>

class Triangle {
private:
    double legA, legB;
public:
    Triangle(double a, double b) : legA(a), legB(b) {}
    
    double getHypotenuse() {
        return std::sqrt(legA * legA + legB * legB);
    }
};

int main() {
    double a, b;
    std::cout << "Введите катеты a и b: ";
    std::cin >> a >> b;
    
    Triangle t(a, b);
    std::cout << "Гипотенуза: " << t.getHypotenuse() << std::endl;
    return 0;
}

    out.close();

}

Задача 2

Модульный:

#include <iostream>
using namespace std;

int tchk(int v, int t) {
    int distance = v * t;
    return distance - (distance / 109) * 109;
}
int main () {
    int v, t;
    cin >> v >> t;
    cout << tchk(v, t);
    return 0;
}

Процедурный:

#include <iostream>
using namespace std;

void Tchk(int v, int t, int &result) {
    int distance = v * t;
    result = distance - (distance / 109) * 109;
}
int main() {
    int v, t, tchk;
    cin >> v >> t;
    Tchk(v, t, tchk);
    cout << tchk;
    return 0;
}
