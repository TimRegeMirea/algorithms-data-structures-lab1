#include <iostream> // подключаем библиотеку для ввода и вывода данных
using namespace std;

int findMissingNumber(int arr[], int n) { // объявляем функцию findMissingNumber, которая принимает массив целых чисел arr, принимает размер массива n, возвращает целое число int - недостающее число.
    int expectedSum = 0; // сюда складываем сумму всех чисел от 0 до n
    int actualSum = 0; // сюда складываем сумму чисел в массиве
    
    for (int i = 0; i <= n; i++) {
        expectedSum += i; // Цикл, который проходит по всем числам от 0 до n включительно и суммирует их в expectedSum. После этого цикла expectedSum будет равен сумме всех чисел, которые должны были быть в массиве (включая недостающее).
    }
    
    for (int i = 0; i < n; i++) {
        actualSum += arr[i]; // // Считаем фактическую сумму элементов массива
    }
    
    
    return expectedSum - actualSum; // // Разница между ожидаемой и фактической суммой и есть недостающее число
}

int main() { // вход в программу
    setlocale(LC_ALL, "Rus"); // устанавливаем русскую локаль
    
    int arr[] = {3, 0, 1, 4, 5, 7, 2}; // Объявляется и инициализируется массив arr с конкретными числами.
    int n = sizeof(arr) / sizeof(arr[0]); // вычисляется кол-во элементов в массиве. sizeof(arr) - размер всего массива. sizeof(arr[0]) - размер одного элемента
    
    cout << "Массив: ";
    for (int i = 0; i < n; i++) {
        cout << arr[i] << " ";
    }
    cout << endl; // Сначала выводится надпись «Массив: », затем циклом печатаются все элементы массива через пробел, а потом endl переводит строку.
    
    int missingNumber = findMissingNumber(arr, n);// Вызывается функция findMissingNumber, передавая ей массив и его размер. Результат сохраняется в переменную MissingNumber.


    cout << "Недостающее число: " << missingNumber << endl;// Вывод недостающего числа
    
    return 0;
}

#include <iostream>

int main() {
    int n;
    std::cout << "Введите n: ";
    std::cin >> n;

    if (n < 1) {
        std::cout << "n должно быть >= 1\n";
        return 0;
    }

    long long current_fact = 1; // здесь будет i! на шаге i
    long long result = 1;       // здесь будет F(n)

    for (int i = 1; i <= n; ++i) { 
        // Сначала обновляем текущий факториал: i! = (i-1)! * i
        current_fact *= i;

        // Теперь умножаем результат на i!
        result *= current_fact;
    }

    std::cout << "F(" << n << ") = " << result << "\n";

    return 0;
}
